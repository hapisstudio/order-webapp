const SPREADSHEET_ID = '1O144AOwQ6_FrxhfrdbZ54Nox7pDC2cWy5illCHvdgTk';

function doGet(e) {
  try {
    const action = e.parameter.action;
    let result = {};

    if (action === 'getBarang') result = { success: true, data: getDaftarBarang() };
    else if (action === 'getRiwayat') result = getRiwayatToko(e.parameter.namaToko);
    else if (action === 'getAllConfig') result = handleGetAllConfig();
    else if (action === 'getConfig') result = handleGetConfig(e.parameter.toko);
    else if (action === 'getUsers') result = handleGetUsers();
    else result = { success: false, message: 'Action GET tidak valid.' };

    return jsonResponse(result);
  } catch (error) {
    return jsonResponse({ success: false, message: error.message });
  }
}

function doPost(e) {
  try {
    const requestData = JSON.parse(e.postData.contents);
    const action = requestData.action;
    let result = {};

    if (action === 'login') result = prosesLogin(requestData.username, requestData.password);
    else if (action === 'simpanPesanan') result = simpanPesanan(requestData.orderData);
    else if (action === 'updateStatus') result = updateStatusSelesai(requestData.idPesanan, requestData.namaToko, requestData.kodeUnik);
    else if (action === 'saveConfig') result = handleSaveConfig(requestData.targetToko, requestData.config);
    else if (action === 'deleteConfig') result = handleDeleteConfig(requestData.targetToko);
    else if (action === 'saveUser') result = handleSaveUser(requestData.user, requestData.mode, requestData.originalUsername);
    else if (action === 'deleteUser') result = handleDeleteUser(requestData.username);
    else result = { success: false, message: 'Action POST tidak valid.' };

    return jsonResponse(result);
  } catch (error) {
    return jsonResponse({ success: false, message: error.message });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function prosesLogin(username, password) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheetUsers = getSheetUsers(ss);
    const data = sheetUsers.getDataRange().getValues();
    const userClean = String(username).trim();
    const passClean = String(password).trim();

    for (let i = 1; i < data.length; i++) {
      const u = String(data[i][0]).trim();
      const p = String(data[i][1]).trim();
      const status = String(data[i][7] || 'Aktif').trim().toLowerCase();

      if (u === userClean && p === passClean) {
        if (status !== 'aktif') return { success: false, message: 'Akun Anda dinonaktifkan.' };
        const role = String(data[i][3] || (u.toLowerCase() === 'admin' ? 'admin' : 'user')).toLowerCase();
        
        return { 
          success: true, 
          namaToko: data[i][2] || u,
          role: role,
          isAdmin: role === 'admin',
          izinGrosir: String(data[i][4]).toLowerCase() === 'true' || data[i][4] === true,
          izinRitel: String(data[i][5]).toLowerCase() === 'true' || data[i][5] === true,
          izinDiskon: String(data[i][6]).toLowerCase() === 'true' || data[i][6] === true
        };
      }
    }
    return { success: false, message: 'Username atau Password salah!' };
  } catch (error) { return { success: false, message: error.message }; }
}

function handleGetUsers() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getSheetUsers(ss);
    const data = sheet.getDataRange().getValues();
    const list = [];
    for (let i = 1; i < data.length; i++) {
      const username = String(data[i][0]).trim();
      if (!username) continue; 
      
      list.push({
        username: username, password: String(data[i][1]), namaToko: String(data[i][2]), role: String(data[i][3] || 'user'),
        izinGrosir: String(data[i][4]).toLowerCase() === 'true' || data[i][4] === true,
        izinRitel: String(data[i][5]).toLowerCase() === 'true' || data[i][5] === true,
        izinDiskon: String(data[i][6]).toLowerCase() === 'true' || data[i][6] === true,
        status: String(data[i][7] || 'Aktif'),
        statusOperasional: String(data[i][9] || ''), judulPengumuman: String(data[i][10] || ''), jamBuka: String(data[i][11] || ''),
        hariLibur: String(data[i][12] || ''), cutOffJam: String(data[i][13] || ''), cutOffDesc: String(data[i][14] || ''),
        catatanGudang: String(data[i][15] || ''), nomorWaAdmin: String(data[i][16] || ''), nomorTelAdmin: String(data[i][17] || '')
      });
    }
    return { success: true, data: list };
  } catch (err) { return { success: false, message: err.message }; }
}

function handleSaveUser(user, mode, originalUsername) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getSheetUsers(ss);
    const data = sheet.getDataRange().getValues();
    const nowStr = new Date().toLocaleString('id-ID');
    const targetUser = mode === 'edit' && originalUsername ? originalUsername : user.username;
    let foundRow = -1;

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim().toLowerCase() === String(targetUser).trim().toLowerCase()) { foundRow = i + 1; break; }
    }

    const rowData = [user.username, user.password, user.namaToko, user.role || 'user', user.izinGrosir || false, user.izinRitel || false, user.izinDiskon || false, user.status || 'Aktif'];

    if (foundRow !== -1) {
      sheet.getRange(foundRow, 1, 1, 8).setValues([rowData]);
      return { success: true, message: 'Akun berhasil diperbarui!' };
    } else {
      sheet.appendRow([...rowData, nowStr, '', '', '', '', '', '', '', '', '']);
      return { success: true, message: 'Akun baru ditambahkan!' };
    }
  } catch (err) { return { success: false, message: err.message }; }
}

function handleDeleteUser(username) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getSheetUsers(ss);
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim().toLowerCase() === String(username).trim().toLowerCase()) {
        sheet.deleteRow(i + 1); return { success: true, message: `Akun dihapus.` };
      }
    }
    return { success: false, message: 'Akun tidak ditemukan.' };
  } catch (err) { return { success: false, message: err.message }; }
}

function handleGetAllConfig() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getSheetUsers(ss);
    const data = sheet.getDataRange().getValues();
    const configMap = {};

    for (let i = 1; i < data.length; i++) {
      const username = String(data[i][0]).trim();
      const namaToko = String(data[i][2]).trim();
      if (!username) continue;

      const key = (username === '__DEFAULT__') ? '__DEFAULT__' : (namaToko || username);
      configMap[key] = {
        statusOperasional: String(data[i][9] || ''), judulPengumuman: String(data[i][10] || ''), jamBuka: String(data[i][11] || ''),
        hariLibur: String(data[i][12] || ''), cutOffJam: String(data[i][13] || ''), cutOffDesc: String(data[i][14] || ''),
        catatanGudang: String(data[i][15] || ''), nomorWaAdmin: String(data[i][16] || ''), nomorTelAdmin: String(data[i][17] || '')
      };
    }
    return { success: true, data: configMap };
  } catch (err) { return { success: false, message: err.message }; }
}

function handleGetConfig(toko) {
  try {
    const resAll = handleGetAllConfig();
    if (!resAll.success) return resAll;
    const all = resAll.data;
    const defaultCfg = all['__DEFAULT__'] || {};
    if (toko && all[toko]) return { success: true, data: Object.assign({}, defaultCfg, all[toko]), isCustom: true };
    return { success: true, data: defaultCfg, isCustom: false };
  } catch (err) { return { success: false, message: err.message }; }
}

function handleSaveConfig(targetToko, config) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getSheetUsers(ss);
    const data = sheet.getDataRange().getValues();
    const targetClean = String(targetToko || '__DEFAULT__').trim().toLowerCase();
    
    let foundRow = -1;
    for (let i = 1; i < data.length; i++) { 
      if (String(data[i][2]).trim().toLowerCase() === targetClean || String(data[i][0]).trim().toLowerCase() === targetClean) { 
        foundRow = i + 1; break; 
      } 
    }
    
    if (foundRow !== -1) {
      const rowData = [
        config.statusOperasional || 'normal', config.judulPengumuman || '', config.jamBuka || '', config.hariLibur || '', 
        config.cutOffJam || '', config.cutOffDesc || '', config.catatanGudang || '', config.nomorWaAdmin || '', config.nomorTelAdmin || ''
      ];
      sheet.getRange(foundRow, 10, 1, 9).setValues([rowData]); 
      return { success: true, message: `Disimpan` }; 
    }

}

function getDaftarBarang() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('Data Stok DA'); 
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];

  const headers = data[0].map(h => String(h).toLowerCase().trim());
  const idxId = headers.indexOf("itemid"); 
  const idxNama = headers.indexOf("itemname"); 
  const idxStok = headers.indexOf("goodqty"); 
  const idxGrosir = headers.indexOf("whprice"); 
  const idxRitel = headers.indexOf("ritailprice"); 
  const idxDiskon = headers.indexOf("maxdisc"); 
  const idxBarcode = headers.indexOf("barcode");

  const items = [];
  function bersihkanAngka(val) { if (!val && val !== 0) return 0; const num = Number(String(val).replace(/[^0-9.-]+/g, "")); return isNaN(num) ? 0 : num; }
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const idVal = idxId !== -1 ? String(row[idxId]) : String(row[0]);
    const namaVal = idxNama !== -1 ? String(row[idxNama]) : String(row[1]);
    if (!idVal && !namaVal) continue; 
    
    items.push({ 
      id: idVal, nama: namaVal, barcode: idxBarcode !== -1 ? String(row[idxBarcode]) : idVal, 
      stok: idxStok !== -1 ? bersihkanAngka(row[idxStok]) : bersihkanAngka(row[2]), 
      hargaGrosir: idxGrosir !== -1 ? bersihkanAngka(row[idxGrosir]) : bersihkanAngka(row[5]), 
      hargaRitel: idxRitel !== -1 ? bersihkanAngka(row[idxRitel]) : bersihkanAngka(row[4]), 
      diskon: idxDiskon !== -1 ? bersihkanAngka(row[idxDiskon]) : bersihkanAngka(row[6]) 
    });
  }
  return items;
}

function simpanPesanan(orderData) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const namaSheetToko = orderData.namaToko;
    let sheetToko = ss.getSheetByName(namaSheetToko);
    if (!sheetToko) { 
      sheetToko = ss.insertSheet(namaSheetToko); 
      sheetToko.appendRow(['ID Pesanan', 'Tanggal', 'Nama Toko', 'ID Barang', 'Nama Barang', 'Jumlah', 'Status', 'Catatan', 'Kode Verifikasi']); 
      sheetToko.getRange("A1:I1").setBackground("#4f46e5").setFontColor("white").setFontWeight("bold"); 
      const ruleStatus = SpreadsheetApp.newDataValidation().requireValueInList(['Pending', 'Diproses', 'Dikirim', 'Selesai', 'Dibatalkan'], true).setAllowInvalid(false).build();
      sheetToko.getRange("G2:G1000").setDataValidation(ruleStatus);
    }
    const now = new Date(); 
    const idPesanan = `ORD-${String(now.getDate()).padStart(2,'0')}${String(now.getMonth()+1).padStart(2,'0')}${String(now.getFullYear()).slice(-2)}-${String(now.getHours()).padStart(2,'0')}${String(now.getMinutes()).padStart(2,'0')}${String(now.getSeconds()).padStart(2,'0')}`;
    const tanggal = now.toLocaleString('id-ID'); 
    const kodeUnikKurir = String(Math.floor(100000 + Math.random() * 900000));
    
    const dataUntukDisimpan = orderData.keranjang.map(item => [ idPesanan, tanggal, orderData.namaToko, item.idBarang, item.namaBarang, Number(item.jumlah), 'Pending', orderData.catatan || '-', kodeUnikKurir ]);
    if (dataUntukDisimpan.length > 0) { sheetToko.getRange(sheetToko.getLastRow() + 1, 1, dataUntukDisimpan.length, dataUntukDisimpan[0].length).setValues(dataUntukDisimpan); }
    return { success: true, message: 'Pesanan dikirim!', idPesanan: idPesanan };
  } catch (error) { return { success: false, message: error.message }; }
}

function getRiwayatToko(namaToko) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheetToko = ss.getSheetByName(namaToko);
    if (!sheetToko) return { success: true, data: [] };

    const data = sheetToko.getDataRange().getValues();
    let orderMap = {};
    for (let i = 1; i < data.length; i++) {
      let id = data[i][0];
      if (!id) continue;
      if (!orderMap[id]) orderMap[id] = { idPesanan: id, tanggal: data[i][1], status: data[i][6] || 'Pending', catatan: data[i][7] || '-', totalItem: 0, totalQty: 0, items: [] };
      orderMap[id].totalItem += 1;
      orderMap[id].totalQty += Number(data[i][5]);
      orderMap[id].items.push({ idBarang: data[i][3], namaBarang: data[i][4], jumlah: data[i][5] });
    }
    return { success: true, data: Object.values(orderMap).reverse() };
  } catch (error) { return { success: false, message: error.message }; }
}

function updateStatusSelesai(idPesanan, namaToko, kodeUnik) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheetToko = ss.getSheetByName(namaToko);
    if (!sheetToko) throw new Error("Sheet toko tidak ditemukan.");
    const data = sheetToko.getDataRange().getValues();
    let kodeValid = false;
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === idPesanan) {
        if (String(data[i][8] || "").trim() === String(kodeUnik).trim()) { kodeValid = true; break; } 
        else { return { success: false, message: 'Kode verifikasi tidak cocok!' }; }
      }
    }
    if (!kodeValid) return { success: false, message: 'Pesanan tidak ditemukan.' };
    for (let i = 1; i < data.length; i++) { if (data[i][0] === idPesanan) { sheetToko.getRange(i + 1, 7).setValue('Selesai'); } }
    return { success: true, message: 'Berhasil' };
  } catch (error) { return { success: false, message: error.message }; }
}
