function doGet(e) {
  try {
    const action = e.parameter.action;
    let result = {};

    if (action === 'getBarang') result = { success: true, data: getDaftarBarang() };
    else if (action === 'getRiwayat') result = getRiwayatToko(e.parameter.namaToko);
    else if (action === 'getAllConfig') result = handleGetAllConfig();
    else if (action === 'getConfig') result = handleGetConfig(e.parameter.toko);
    else if (action === 'getUsers') result = handleGetUsers();
    else result = { success: false, message: 'Action GET tidak valid.' };

    return jsonResponse(result);
  } catch (error) {
    return jsonResponse({ success: false, message: error.message });
  }
}

function doPost(e) {
  try {
    const requestData = JSON.parse(e.postData.contents);
    const action = requestData.action;
    let result = {};

    if (action === 'login') result = prosesLogin(requestData.username, requestData.password);
    else if (action === 'simpanPesanan') result = simpanPesanan(requestData.orderData);
    else if (action === 'updateStatus') result = updateStatusSelesai(requestData.idPesanan, requestData.namaToko, requestData.kodeUnik);
    else if (action === 'saveConfig') result = handleSaveConfig(requestData.targetToko, requestData.config);
    else if (action === 'deleteConfig') result = handleDeleteConfig(requestData.targetToko);
    else if (action === 'saveUser') result = handleSaveUser(requestData.user, requestData.mode, requestData.originalUsername);
    else if (action === 'deleteUser') result = handleDeleteUser(requestData.username);
    else result = { success: false, message: 'Action POST tidak valid.' };

    return jsonResponse(result);
  } catch (error) {
    return jsonResponse({ success: false, message: error.message });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
